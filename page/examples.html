Hello World

Your first Braw program.

import "stdinout.braw"
                    
fn main() -> int {
    print_string("Hello, world!\n");
    return 0;
}

Compile it using this command.

./brawc main.braw --assemble --link

Variables

Declaring and using variables in Braw.

let x: int = 10;
let y = x * 2;

Functions

How to declare and use functions.

fn times2(n: int) -> int {
    return n * 2;
}
You can also declare single line functions like this: 
fn times2(n: int) -> int: return n * 2;

If statements

let n = 5;

if (n > 3):
    print_string("Big\n");
else:
    print_string("Small\n");

While loops

let n = 0;

while (n < 10) {
    print_string("Hello, world!\n");
    n = n + 1;
}

You can also run do-while loops:

do while (n < 10) {
    print_string("Hello, world!\n");
    n = n + 1;
}

For loops

for (let i = 0; i < 10; i = i + 1):
    print_string("Hello, world!\n");

Macros

Macros in Braw are defined using the define keyword and invoked using the $ symbol.

define CONST: 5;

fn wow() -> void {
    let a: int = $CONST;
}

Macros can take parameters:

define CONST(param): 5 + #param;

fn wow() -> void {
    let a: int = $CONST(5);
}

Full example

struct MV {
    a: int;
    b: int;
};

define MAC(type) {
    $foreach(var : #type) {
        $make_function($concat($concat(#type, #"_"), $concat(#"get_", #var.name)), #var, {$make_variable(#"v", #type);}) {
            return $make_dot(v, #var.name);
        }
    }
}

$MAC(#<MV>);

fn m_macro() -> int {
    let v: MV; v.a = 1; v.b = 1;
    return MV_get_a(v) + MV_get_b(v);
}
